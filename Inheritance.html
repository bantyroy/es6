Inheritance


class FrontendTechnology {
    getReactJS() {
        console.warn("ReactJS" );
    }

    getAngular = () => {
        console.warn("Angular");
    }

    getVueJS() {
        console.warn("VueJS" );
    }
}

class Website extends FrontendTechnology {
    constructor(name) {
        super(); // Call the parent class constructor
        this.name = name;
    }

    getName() {
        console.warn("Name: ", this.name);
    }    
}

const item = new Website("Facebook");
item.getName(); // Output: Name: Facebook
item.getReactJS(); // Output: ReactJS
