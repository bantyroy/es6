Lexical Scope for This


let data={
    list:'friend',
    names:['Bruce','Peter','Sam'],    
    getFriendsNormal:function()
    {
        let _this = this; // As function creates its own scope, we need to store the reference of 'this' in a variable
        this.names.map(function(item){
            console.warn(_this.list,item)
        })
    },
    getFriends:function()
    {
        // Arrow functions do not have their own 'this' context, they inherit it from the parent scope
        this.names.map((item)=>{
            console.warn(this.list,item)
        })
    }
}

data.getFriends();
data.getFriendsNormal();
