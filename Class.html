Classes


class Person {
    topic = "ES6 Classes";
    constructor(name) {
        this.name = name;
        console.warn("Constructor", this.topic);
    }

    getName() {
        console.warn("Normal", this.name);
    }

    getNameArrow = () => {
        console.warn("Arrow", this.name);
    }
}

const item = new Person("J Samanta");
item.getName();
item.getNameArrow();
